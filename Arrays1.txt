5).ARRAYS TO PRINT MAX & MIN VALUE OF THE GIVEN NUMBERS(RUN TIME)?

using System;
class Program
{
    static void Main()
    {
        int min = 0; int max = 0;                                 
        Console.WriteLine("enter the size of an array: ");
        int size = int.Parse(Console.ReadLine());
        int[] ar = new int[size];
        for (int i = 0; i < size; i++)
        {
            Console.WriteLine("enter numbers: ");

            ar[i] = int.Parse(Console.ReadLine());
        }
        min = ar[0];
        max = ar[0];
        for (int i = 0; i < size; i++)
        {
            if (min > ar[i])
            {
                min = ar[i];
            }
            if (max < ar[i])
            {
                max = ar[i];
            }
        }
        Console.WriteLine("minmum value is: " + min);
        Console.WriteLine("maxmum value is: " + max);
    }
}

10).ARRAYS TO PRINT REVERS OF THE STRING?

using System;
class Program
{                                         
    static void Main()
    {
        Console.WriteLine("enter a string: ");
        string s = Console.ReadLine();
        string r = "";
        for (int i = s.Length - 1; i >= 0; i--)
        {
            r += s[i];
        }
        Console.WriteLine("Riversed string: " + r);
    }
}
