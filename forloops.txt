12).FORLOOP TO PRINT THE MIN & MAX VALUE OF THE GIVEN NUMBERS AT RUNTIME?

using System;

class Program
{
    static void Main()
    {
        Console.Write("Enter how many numbers: ");
        int n = int.Parse(Console.ReadLine());

        int min =int.MaxValue;   // start with largest possible value
        int max =int.MinValue;   // start with smallest possible value

        Console.WriteLine("Enter the numbers:");
        for (int i = 0; i < n; i++)
        {
            int num = int.Parse(Console.ReadLine());
             
            if (num < min)
            {
                min = num;   // update min
            }
            if (num > max)
            {
                max = num;   // update max
            }
        }

        Console.WriteLine("Minimum number = " + min);
        Console.WriteLine("Maximum number = " + max);
    }
}
